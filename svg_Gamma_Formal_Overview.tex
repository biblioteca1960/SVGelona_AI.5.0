\section{Axiomatic Foundations}

\end{axiom}

\begin{axiom}[Angular Deviation]

\begin{axiom}[Energy]

$E(\vec{z}) \ge 0$, and $E(\vec{z}) = 0$ iff $\vec{z} \in L_c$.

\begin{axiom}[Symmetry]
